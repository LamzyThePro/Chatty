const ITERATIONS = 100000;
const SESSION_DAYS = 30;
const INVITE_ALPHABET = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
const MAX_APP_BYTES = 1800000;
const APP_MARKER = 'name="application-name" content="Chatty"';

const cors = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Methods": "GET, POST, DELETE, OPTIONS",
  "Access-Control-Allow-Headers": "Content-Type, Authorization",
  "Content-Type": "application/json"
};

function json(data, status = 200) {
  return new Response(JSON.stringify(data), { status, headers: cors });
}

function randomId(bytes = 24) {
  const data = crypto.getRandomValues(new Uint8Array(bytes));
  return btoa(String.fromCharCode(...data))
    .replace(/\+/g, "-").replace(/\//g, "_").replace(/=/g, "");
}

function randomInviteCode() {
  const bytes = crypto.getRandomValues(new Uint8Array(6));
  return Array.from(bytes, byte => INVITE_ALPHABET[byte & 31]).join("");
}

async function hash(value) {
  const data = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(value));
  return btoa(String.fromCharCode(...new Uint8Array(data)));
}

async function sha256Hex(bytes) {
  const digest = await crypto.subtle.digest("SHA-256", bytes);
  return Array.from(new Uint8Array(digest), b => b.toString(16).padStart(2, "0")).join("");
}

function constantTimeEqual(a, b) {
  if (typeof a !== "string" || typeof b !== "string" || a.length !== b.length) return false;
  let diff = 0;
  for (let i = 0; i < a.length; i++) diff |= a.charCodeAt(i) ^ b.charCodeAt(i);
  return diff === 0;
}

function nextPatchVersion(value) {
  const match = String(value || "1.0.0").match(/^(\d+)\.(\d+)\.(\d+)(?:[-_.].*)?$/);
  if (!match) return "1.0.0";
  return `${Number(match[1])}.${Number(match[2])}.${Number(match[3]) + 1}`;
}

function compareVersions(a, b) {
  const parse = v => {
    const m = String(v).match(/^(\d+(?:\.\d+)*)(?:[-_.]?(.+))?$/);
    return m
      ? { nums: m[1].split(".").map(Number), pre: m[2] || "" }
      : { nums: [0], pre: String(v) };
  };

  const x = parse(a), y = parse(b), len = Math.max(x.nums.length, y.nums.length);

  for (let i = 0; i < len; i++) {
    const d = (x.nums[i] || 0) - (y.nums[i] || 0);
    if (d) return d > 0 ? 1 : -1;
  }

  if (x.pre === y.pre) return 0;
  if (!x.pre) return 1;
  if (!y.pre) return -1;
  return x.pre < y.pre ? -1 : 1;
}

async function passwordHash(password) {
  const salt = crypto.getRandomValues(new Uint8Array(16));
  const key = await crypto.subtle.importKey(
    "raw", new TextEncoder().encode(password), "PBKDF2", false, ["deriveBits"]
  );
  const bits = await crypto.subtle.deriveBits(
    { name: "PBKDF2", salt, iterations: ITERATIONS, hash: "SHA-256" },
    key, 256
  );
  const encode = bytes => btoa(String.fromCharCode(...new Uint8Array(bytes)));
  return `pbkdf2$${ITERATIONS}$${encode(salt)}$${encode(bits)}`;
}

function storedPasswordIterations(stored) {
  if (typeof stored !== "string") return 0;
  const parts = stored.split("$");
  const n = Number(parts[1]);
  return parts[0] === "pbkdf2" && Number.isInteger(n) ? n : 0;
}

async function verifyPassword(password, stored) {
  try {
    const [, iterationsText, saltText, expected] = stored.split("$");
    const iterations = Number(iterationsText);

    if (!Number.isInteger(iterations) || iterations < 10000 || iterations > 1000000) {
      return false;
    }

    const decode = text => Uint8Array.from(atob(text), c => c.charCodeAt(0));
    const salt = decode(saltText);
    const expectedBytes = decode(expected);

    if (salt.length !== 16 || expectedBytes.length !== 32) return false;

    const key = await crypto.subtle.importKey(
      "raw", new TextEncoder().encode(password), "PBKDF2", false, ["deriveBits"]
    );
    const bits = await crypto.subtle.deriveBits(
      { name: "PBKDF2", salt, iterations, hash: "SHA-256" }, key, 256
    );

    return constantTimeEqual(
      btoa(String.fromCharCode(...new Uint8Array(bits))),
      expected
    );
  } catch {
    return false;
  }
}

const RATE_LIMIT_POLICIES = Object.freeze({
  register: { default: 8, min: 1, max: 20 },
  login: { default: 10, min: 1, max: 30 },
  "user-search": { default: 30, min: 1, max: 120 },
  "send-message": { default: 60, min: 10, max: 240 },
  "group-create": { default: 5, min: 1, max: 20 },
  "group-member": { default: 30, min: 5, max: 120 }
});

async function getRateLimitValue(env, label) {
  const policy = RATE_LIMIT_POLICIES[label];
  if (!policy) return 10;
  const stored = await getSetting(env, `rate_limit_${label}`, String(policy.default));
  return boundedInt(stored, policy.default, policy.min, policy.max);
}

async function incrementRateBucket(env, key, now, windowMs) {
  const result = await env.DB.prepare(`
    INSERT INTO rate_limits (rate_key, window_start, attempts) VALUES (?, ?, 1)
    ON CONFLICT(rate_key) DO UPDATE SET
      attempts = CASE WHEN window_start <= ? THEN 1 ELSE attempts + 1 END,
      window_start = CASE WHEN window_start <= ? THEN ? ELSE window_start END
    RETURNING attempts
  `).bind(key, now, now - windowMs, now - windowMs, now).first();

  return Number(result?.attempts || 0);
}

let rateCleanupTick = 0;

async function rateLimit(request, env, label, identity = null) {
  const limit = await getRateLimitValue(env, label);
  const ip = request.headers.get("CF-Connecting-IP") || "unknown";
  const now = Date.now(), windowMs = 15 * 60 * 1000;

  const ipKey = await hash(`v2:${label}:ip:${ip}`);
  const ipAttempts = await incrementRateBucket(env, ipKey, now, windowMs);

  let identityAttempts = 0;

  if (typeof identity === "string" && identity.length) {
    const normalized = identity.trim().toLowerCase().slice(0, 160);
    const identityKey = await hash(`v2:${label}:identity:${normalized}`);
    identityAttempts = await incrementRateBucket(env, identityKey, now, windowMs);
  }

  rateCleanupTick = (rateCleanupTick + 1) % 100;

  if (rateCleanupTick === 0) {
    await env.DB.prepare(`
      DELETE FROM rate_limits WHERE rate_key IN (
        SELECT rate_key FROM rate_limits WHERE window_start < ? LIMIT 250
      )
    `).bind(now - 24 * 60 * 60 * 1000).run();
  }

  return ipAttempts > 0 && ipAttempts <= limit &&
    (identity === null || (identityAttempts > 0 && identityAttempts <= limit));
}

async function getUser(request, env) {
  const header = request.headers.get("Authorization") || "";
  const token = header.startsWith("Bearer ") ? header.slice(7) : "";
  if (!token || token.length > 200) return null;

  const tokenHash = await hash(token);

  const user = await env.DB.prepare(`
    SELECT u.id, u.username, u.role
    FROM sessions s JOIN users u ON u.id = s.user_id
    WHERE s.token_hash = ? AND s.expires_at > ?
  `).bind(tokenHash, Date.now()).first();

  if (!user) return null;

  const ban = await env.DB.prepare("SELECT user_id FROM bans WHERE user_id = ?")
    .bind(user.id).first();

  return ban ? null : user;
}

function publicUser(user) {
  if (!user) return null;
  const privileged = ["owner", "moderator", "admin"].includes(user.role);

  return {
    id: user.id,
    username: user.username,
    role: privileged ? "admin" : "user",
    accessLevel: user.role
  };
}

async function createSession(env, userId) {
  const token = randomId(32);

  await env.DB.prepare(`
    INSERT INTO sessions (token_hash, user_id, expires_at) VALUES (?, ?, ?)
  `).bind(await hash(token), userId, Date.now() + SESSION_DAYS * 86400000).run();

  return token;
}

async function readBody(request, maxChars = 10000) {
  const maxBytes = maxChars * 4;
  const declaredHeader = request.headers.get("Content-Length");

  if (declaredHeader !== null) {
    const declared = Number(declaredHeader);
    if (!Number.isSafeInteger(declared) || declared < 0) {
      throw new Error("Invalid request length");
    }
    if (declared > maxBytes) throw new Error("Request too large");
  }

  if (!request.body) throw new Error("Invalid JSON body");

  const reader = request.body.getReader();
  const chunks = [];
  let total = 0;

  try {
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;

      total += value.byteLength;
      if (total > maxBytes) {
        await reader.cancel();
        throw new Error("Request too large");
      }

      chunks.push(value);
    }
  } finally {
    reader.releaseLock();
  }

  const bytes = new Uint8Array(total);
  let offset = 0;

  for (const chunk of chunks) {
    bytes.set(chunk, offset);
    offset += chunk.byteLength;
  }

  let text;

  try {
    text = new TextDecoder("utf-8", { fatal: true }).decode(bytes);
  } catch {
    throw new Error("Invalid JSON body");
  }

  if (text.length > maxChars) throw new Error("Request too large");

  let parsed;

  try {
    parsed = JSON.parse(text);
  } catch {
    throw new Error("Invalid JSON body");
  }

  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
    throw new Error("Invalid JSON body");
  }

  return parsed;
}

function validUsername(username) {
  return typeof username === "string" && /^[a-zA-Z0-9_-]{3,20}$/.test(username);
}

function validPassword(password) {
  return typeof password === "string" && password.length >= 10 && password.length <= 128;
}

function isOwner(user) {
  return user?.role === "owner";
}

function isPrivileged(user) {
  return user?.role === "owner" || user?.role === "moderator" || user?.role === "admin";
}

async function adminOnly(request, env) {
  const user = await getUser(request, env);
  return isPrivileged(user) ? user : null;
}

function ownerOnly(user) {
  return isOwner(user);
}

async function ensureGroupSchema(env) {
  if (groupSchemaReady) return;
  await env.DB.batch([
    env.DB.prepare(`CREATE TABLE IF NOT EXISTS group_details (
      conversation_id TEXT PRIMARY KEY,
      name TEXT NOT NULL,
      created_by TEXT NOT NULL,
      created_at INTEGER NOT NULL,
      FOREIGN KEY (conversation_id) REFERENCES conversations(id),
      FOREIGN KEY (created_by) REFERENCES users(id)
    )`),
    env.DB.prepare(`CREATE TABLE IF NOT EXISTS group_admins (
      conversation_id TEXT NOT NULL,
      user_id TEXT NOT NULL,
      assigned_at INTEGER NOT NULL,
      assigned_by TEXT NOT NULL,
      PRIMARY KEY (conversation_id, user_id),
      FOREIGN KEY (conversation_id) REFERENCES conversations(id),
      FOREIGN KEY (user_id) REFERENCES users(id),
      FOREIGN KEY (assigned_by) REFERENCES users(id)
    )`),
    env.DB.prepare(`CREATE TABLE IF NOT EXISTS message_reactions (
      message_id INTEGER NOT NULL,
      user_id TEXT NOT NULL,
      reaction TEXT NOT NULL,
      created_at INTEGER NOT NULL,
      PRIMARY KEY (message_id, user_id, reaction),
      FOREIGN KEY (message_id) REFERENCES messages(id),
      FOREIGN KEY (user_id) REFERENCES users(id)
    )`),
    env.DB.prepare("CREATE INDEX IF NOT EXISTS idx_group_admins_user ON group_admins(user_id)"),
    env.DB.prepare("CREATE INDEX IF NOT EXISTS idx_reactions_message ON message_reactions(message_id)")
  ]);
  const columns = await env.DB.prepare("PRAGMA table_info(messages)").all();
  if (!(columns.results || []).some(c => c.name === "reply_to_message_id")) {
    try {
      await env.DB.prepare("ALTER TABLE messages ADD COLUMN reply_to_message_id INTEGER REFERENCES messages(id)").run();
    } catch (error) {
      // Another Worker isolate may have completed the same additive migration.
      const refreshed = await env.DB.prepare("PRAGMA table_info(messages)").all();
      const exists = (refreshed.results || []).some(c => c.name === "reply_to_message_id");
      if (!exists) throw error;
    }
  }
  groupSchemaReady = true;
}

async function arcadeGitHubRequest(env, path, options = {}) {
  if (!env.GITHUB_TOKEN) throw new Error("GitHub storage is not configured. Add the GITHUB_TOKEN Worker secret.");
  const response = await fetch("https://api.github.com" + path, {
    ...options,
    headers: {
      "Authorization": "Bearer " + env.GITHUB_TOKEN,
      "Accept": options.accept || "application/vnd.github+json",
      "X-GitHub-Api-Version": "2022-11-28",
      ...(options.headers || {})
    }
  });
  if (!response.ok) {
    let message = "";
    try { message = (await response.json()).message || ""; } catch (_) {}
    throw new Error("GitHub API error (" + response.status + ")" + (message ? ": " + message : ""));
  }
  return response;
}

async function getArcadeRelease(env) {
  if (!env.GITHUB_TOKEN) throw new Error("GitHub storage is not configured. Add the GITHUB_TOKEN Worker secret.");
  const base = "/repos/" + ARCADE_GITHUB_OWNER + "/" + ARCADE_GITHUB_REPO;
  const headers = {
    "Authorization": "Bearer " + env.GITHUB_TOKEN,
    "Accept": "application/vnd.github+json",
    "X-GitHub-Api-Version": "2022-11-28"
  };
  let response = await fetch("https://api.github.com" + base + "/releases/tags/" + ARCADE_GITHUB_TAG, { headers });
  if (response.status === 404) {
    response = await fetch("https://api.github.com" + base + "/releases", {
      method: "POST", headers: { ...headers, "Content-Type": "application/json" },
      body: JSON.stringify({ tag_name: ARCADE_GITHUB_TAG, name: "Chatty Arcade Library", body: "Shared game files used by Chatty Arcade." })
    });
    if (response.status === 422) response = await fetch("https://api.github.com" + base + "/releases/tags/" + ARCADE_GITHUB_TAG, { headers });
  }
  if (!response.ok) {
    let message = "";
    try { message = (await response.json()).message || ""; } catch (_) {}
    throw new Error("Could not access Chatty Arcade release (" + response.status + ")" + (message ? ": " + message : ""));
  }
  return response.json();
}

      }
    }

  await env.DB.prepare("CREATE INDEX IF NOT EXISTS idx_arcade_games_added ON arcade_games(added_at DESC)").run();

async function ensureAdminSchema(env) {
  if (adminSchemaReady) return;

  const inviteInfo = await env.DB.prepare("PRAGMA table_info(invites)").all();
  const inviteColumns = new Set((inviteInfo.results || []).map(c => c.name));

  if (!inviteColumns.has("id")) {
    await env.DB.prepare("ALTER TABLE invites ADD COLUMN id TEXT").run();
  }

  await env.DB.prepare(`
    UPDATE invites SET id = lower(hex(randomblob(16)))
    WHERE id IS NULL OR id = ''
  `).run();

  await env.DB.prepare(
    "CREATE UNIQUE INDEX IF NOT EXISTS idx_invites_id ON invites(id)"
  ).run();

  await env.DB.prepare(
    "CREATE INDEX IF NOT EXISTS idx_sessions_user_expiry ON sessions(user_id, expires_at)"
  ).run();

  await env.DB.prepare(
    "CREATE INDEX IF NOT EXISTS idx_users_username ON users(username)"
  ).run();

  // Promote one legacy admin only when no owner already exists.
  // If an owner exists, all legacy admin accounts become moderators instead.
  await env.DB.prepare(`
    UPDATE users SET role = 'owner'
    WHERE role = 'admin'
      AND NOT EXISTS (SELECT 1 FROM users WHERE role = 'owner')
      AND id = (
        SELECT id FROM users WHERE role = 'admin'
        ORDER BY created_at ASC, id ASC LIMIT 1
      )
  `).run();

  await env.DB.prepare(
    "UPDATE users SET role = 'moderator' WHERE role = 'admin'"
  ).run();

  await env.DB.prepare(`
    INSERT OR IGNORE INTO app_settings (key, value)
    VALUES ('maintenance_mode', 'false')
  `).run();

  await env.DB.prepare(`
    INSERT OR IGNORE INTO app_settings (key, value)
    VALUES ('maintenance_message', 'Chatty is temporarily unavailable for maintenance.')
  `).run();

  adminSchemaReady = true;
}

function boundedInt(value, fallback, min, max) {
  const n = Number(value);
  return Number.isInteger(n) && n >= min && n <= max ? n : fallback;
}

function escapeLike(value) {
  return value.replace(/[\\%_]/g, ch => "\\" + ch);
}

export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const path = url.pathname;

    try {
      if (request.method === "OPTIONS") {
        return new Response(null, { headers: cors });
      }

      if (!env.DB) return json({ error: "Database unavailable" }, 503);

      if (path === "/api/health" && request.method === "GET") {
        await env.DB.prepare("SELECT 1").first();
        return json({ status: "online", database: "connected", app: "Chatty" });
      }

      if (path === "/api/config" && request.method === "GET") {
        // Public read-only endpoint: never run database migrations here.
        return json({
          registrationMode: await getRegistrationMode(env),
          maintenanceMode: (await getSetting(env, "maintenance_mode", "false")) === "true",
          maintenanceMessage: await getSetting(
            env, "maintenance_message",
            "Chatty is temporarily unavailable for maintenance."
          )
        });
      }

      if (path === "/api/update" && request.method === "GET") {
        // Public read-only endpoint. Schema changes are performed only by
        // authenticated Owner update-management routes.
        const update = await env.DB.prepare(`
          SELECT id, version, release_notes, download_url, force_update, sha256, created_at
          FROM app_updates ORDER BY id DESC LIMIT 1
        `).first();

        if (!update) return json({ update: null });

        return json({
          update: {
            id: update.id,
            version: update.version,
            releaseNotes: update.release_notes,
            downloadUrl: update.download_url,
            forceUpdate: Boolean(update.force_update),
            sha256: update.sha256 || null,
            publishedAt: update.created_at
          }
        });
      }

      const fileMatch = path.match(/^\/app\/file\/(\d{1,10})$/);

      if (fileMatch && request.method === "GET") {
        // Public download endpoint: do not run migrations during downloads.
        const file = await env.DB.prepare(
          "SELECT html, sha256 FROM app_files WHERE id = ?"
        ).bind(Number(fileMatch[1])).first();

        if (!file) return json({ error: "File not found" }, 404);

        return new Response(file.html, {
          headers: {
            "Content-Type": "text/html; charset=utf-8",
            "Content-Disposition": 'attachment; filename="Chatty.html"',
            "Access-Control-Allow-Origin": "*",

            }

            const declared = body.html.match(/const APP_VERSION = ['"]([^'"]{1,32})['"]/);
            if (!declared) return json({ error: "The uploaded file has no APP_VERSION declaration" }, 400);
            if (autoVersionRequested && !/^\d+\.\d+\.\d+(?:[-_.].*)?$/.test(declared[1])) {
              return json({ error: "The uploaded file has an invalid APP_VERSION; use major.minor.patch format" }, 400);
            }

            const versionRows = await env.DB.prepare("SELECT version FROM app_updates").all();
            let highestPublishedVersion = null;
            for (const row of versionRows.results || []) {
              if (typeof row.version === "string" &&
                  (!highestPublishedVersion || compareVersions(row.version, highestPublishedVersion) > 0)) {
                highestPublishedVersion = row.version;
              }
            }

            if (autoVersionRequested) {
              const baseVersion = highestPublishedVersion && compareVersions(highestPublishedVersion, declared[1]) > 0
                ? highestPublishedVersion
                : declared[1];
              body.version = nextPatchVersion(baseVersion);
            }

            body.html = body.html.replace(/const APP_VERSION = ['"][^'"]{1,32}['"];?/, `const APP_VERSION = '${body.version}';`);
            const releaseDate = new Date(now).toISOString().slice(0, 10);
            if (/const APP_RELEASE_DATE = ['"][^'"]*['"]/.test(body.html)) {
              body.html = body.html.replace(/const APP_RELEASE_DATE = ['"][^'"]*['"];/, `const APP_RELEASE_DATE = '${releaseDate}';`);
            } else {
              body.html = body.html.replace(/(const APP_VERSION = '[^']+';)/, `$1\nconst APP_RELEASE_DATE = '${releaseDate}';`);
            }
            body.html = body.html.replace(/<title>Chatty v[^<]*<\/title>/i, `<title>Chatty v${body.version} · ${releaseDate}</title>`);

            if (highestPublishedVersion && compareVersions(body.version, highestPublishedVersion) <= 0) {
              return json({
                error: `Version must be higher than the highest previously published version ${highestPublishedVersion}. To go back to an older version, use Republish in the update history.`
              }, 400);
            }

            // Hash the exact HTML stored and served, after version/date stamping.
            const bytes = new TextEncoder().encode(body.html);
            if (bytes.length > MAX_APP_BYTES) {
              return json({ error: "The stamped app file is too large" }, 413);
            }
            const sha = await sha256Hex(bytes);
            const prefix = `${url.origin}/app/file/`;

            await env.DB.batch([
              env.DB.prepare(`
                INSERT INTO app_files (version, sha256, html, created_at)
                VALUES (?, ?, ?, ?)
              `).bind(body.version, sha, body.html, now),

              env.DB.prepare(`
                INSERT INTO app_updates (
                  version, release_notes, download_url, force_update,
                  published_by, created_at, sha256, file_id
                )
                VALUES (?, ?, ? || last_insert_rowid(), ?, ?, ?, ?, last_insert_rowid())
              `).bind(
                body.version, body.releaseNotes, prefix,
                force, admin.id, now, sha
              )
            ]);

            await audit(env, admin.id, "update_published", body.version);

            return json({
              success: true,
              version: body.version,
              sha256: sha,
              publishedAt: now
            }, 201);
          }

          if (autoVersionRequested) {
            return json({ error: "Automatic versioning requires uploading the Chatty HTML file" }, 400);
          }
          if (typeof body.downloadUrl !== "string" || body.downloadUrl.length > 2048) {
            return json({ error: "Send the app file, or a valid download URL" }, 400);
          }

          let downloadUrl;

          try {
            downloadUrl = new URL(body.downloadUrl);
          } catch {
            return json({ error: "Invalid download URL" }, 400);
          }

          if (downloadUrl.protocol !== "https:") {
            return json({ error: "Download URL must use HTTPS" }, 400);
          }

          const legacySha =
            typeof body.sha256 === "string" && /^[a-f0-9]{64}$/.test(body.sha256)
              ? body.sha256
              : null;

          await env.DB.prepare(`
            INSERT INTO app_updates (
              version, release_notes, download_url, force_update,
              published_by, created_at, sha256, file_id
            )
            VALUES (?, ?, ?, ?, ?, ?, ?, NULL)
          `).bind(
            body.version, body.releaseNotes, downloadUrl.href,
            force, admin.id, now, legacySha
          ).run();

          await audit(env, admin.id, "update_published", body.version);

          return json({ success: true, version: body.version, publishedAt: now }, 201);
        }

        if (path === "/api/admin/updates" && request.method === "GET") {
          if (!owner) return json({ error: "Owner access required" }, 403);

          await ensureSchema(env);

          const result = await env.DB.prepare(`
            SELECT id, version, release_notes, download_url, force_update,
              published_by, created_at, sha256, file_id
            FROM app_updates ORDER BY id DESC LIMIT 50
          `).all();

          return json({ updates: result.results });
        }

        if (path === "/api/admin/update/rollback" && request.method === "POST") {
          if (!owner) return json({ error: "Owner access required" }, 403);

          await ensureSchema(env);

          const body = await readBody(request);
          const updateId = Number(body.updateId);

          if (!Number.isInteger(updateId) || updateId < 1) {
            return json({ error: "Invalid update ID" }, 400);
          }

          const previous = await env.DB.prepare(`
            SELECT version, release_notes, download_url, force_update, sha256, file_id
            FROM app_updates WHERE id = ?
          `).bind(updateId).first();

          if (!previous) return json({ error: "Update not found" }, 404);

          if (previous.file_id) {
            const file = await env.DB.prepare(
              "SELECT sha256 FROM app_files WHERE id = ?"
            ).bind(previous.file_id).first();

            if (!file || !previous.sha256 || file.sha256 !== previous.sha256) {
              return json({
                error: "The stored app file is missing or its checksum does not match; rollback was stopped"
              }, 409);
            }
          }

          const now = Date.now();

          await env.DB.prepare(`
            INSERT INTO app_updates (
              version, release_notes, download_url, force_update,
              published_by, created_at, sha256, file_id
            )
            VALUES (?, ?, ?, ?, ?, ?, ?, ?)
          `).bind(
            previous.version,
            `Rollback: ${previous.release_notes || ""}`.slice(0, 2000),
            previous.download_url,
            previous.force_update,
            admin.id,
            now,
            previous.sha256,
            previous.file_id
          ).run();

          await audit(env, admin.id, "update_rollback", String(updateId));

          return json({
            success: true,
            version: previous.version,
            rolledBackFrom: updateId
          }, 201);
        }

        return json({ error: "Admin endpoint not found" }, 404);
      }

      return json({ error: "Endpoint not found" }, 404);
    } catch (error) {
      console.error("Chatty Worker error:", error);

      if (error.message?.includes("Request too large")) {
        return json({ error: "Request too large" }, 413);
      }

      if (error.message?.includes("Invalid JSON body") ||
          error.message?.includes("Invalid request length")) {
        return json({ error: "Invalid JSON request body" }, 400);
      }

      if (error.message?.includes("UNIQUE constraint failed")) {
        return json({ error: "Username already taken" }, 409);
      }

      return json({ error: "Internal server error" }, 500);
    }
  }
};
